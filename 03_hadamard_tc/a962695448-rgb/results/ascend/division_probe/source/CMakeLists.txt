cmake_minimum_required(VERSION 3.16)
project(ascend_div_probe LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(ASCEND_CANN_PACKAGE_PATH "/usr/local/Ascend/cann-9.0.0" CACHE PATH "Actual CANN installation")
set(SOC_VERSION "Ascend910B1" CACHE STRING "Real target verified with aclrtGetSocName")
set(RUN_MODE "npu" CACHE STRING "Real NPU execution only")
option(ENABLE_SCALAR_DIV "Also compile aicore C++ float /; independent vector-only build remains available" OFF)
if(NOT RUN_MODE STREQUAL "npu")
    message(FATAL_ERROR "Only RUN_MODE=npu is accepted; CPU/simulator is not a precision result")
endif()
set(ASCENDC_CMAKE_FILE "${ASCEND_CANN_PACKAGE_PATH}/aarch64-linux/tikcpp/ascendc_kernel_cmake/ascendc.cmake"
    CACHE FILEPATH "Kernel Launch CMake entry from installed SDK")
if(NOT EXISTS "${ASCENDC_CMAKE_FILE}")
    message(FATAL_ERROR "Missing Ascend C CMake entry: ${ASCENDC_CMAKE_FILE}")
endif()
include("${ASCENDC_CMAKE_FILE}")
if(ENABLE_SCALAR_DIV)
    set(PROBE_SCALAR_DIV 1)
else()
    set(PROBE_SCALAR_DIV 0)
endif()
ascendc_library(ascend_div_probe_kernel STATIC kernel.cpp)
ascendc_compile_definitions(ascend_div_probe_kernel PRIVATE PROBE_SCALAR_DIV=${PROBE_SCALAR_DIV})
add_executable(ascend_div_probe main.cpp)
target_compile_definitions(ascend_div_probe PRIVATE PROBE_SCALAR_DIV=${PROBE_SCALAR_DIV})
# Host oracle must use FP32 /, not reciprocal approximation or contraction.
target_compile_options(ascend_div_probe PRIVATE -fno-fast-math -ffp-contract=off -frounding-math)
target_link_libraries(ascend_div_probe PRIVATE ascend_div_probe_kernel host_intf_pub)
